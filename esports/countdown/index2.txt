function showScreen(name) {
  $('publicScreen').style.display = 'none';
  $('loginScreen').style.display  = 'none';
  $('adminScreen').style.display  = 'none';

  if (name === 'public') {
    $('publicScreen').style.display = 'block';
    window.location.hash = '';
  } else if (name === 'login') {
    $('loginScreen').style.display = 'flex';
    window.location.hash = 'admin';
    setTimeout(() => $('pwInput').focus(), 100);
  } else if (name === 'admin') {
    $('adminScreen').style.display = 'block';
    window.location.hash = 'admin';
    renderAdmin();
  }
}

window.goAdmin  = () => loggedIn ? showScreen('admin') : showScreen('login');

  if (window.location.hash === '#admin') {
    loggedIn ? showScreen('admin') : showScreen('login');
  } else {
    showScreen('public');
  }
}
window.addEventListener('hashchange', handleHash);

onValue(ref(db, 'schedules'), snapshot => {
  const data = snapshot.val();
  allItems = data
    ? Object.entries(data)
        .map(([id, v]) => ({ id, ...v }))
        .sort((a, b) => a.targetTime - b.targetTime)
    : [];
  renderPublic();
  if (loggedIn) renderAdmin();
});

// ──────────────────────────────────────
// 유틸
// ──────────────────────────────────────
function fmtDate(ts) {
  const d = new Date(ts);
  const day = ['일','월','화','수','목','금','토'][d.getDay()];
  return `${d.getMonth()+1}/${d.getDate()}(${day})`;
}
function fmtTime(ts) {
  const d = new Date(ts);
  return `${pad(d.getHours())}:${pad(d.getMinutes())}`;
}
function getDayKey(ts) {
  const d = new Date(ts);
  return `${d.getFullYear()}-${pad(d.getMonth()+1)}-${pad(d.getDate())}`;
}
function fmtCd(ms) {
  if (ms <= 0) return null;
  const s = Math.floor(ms / 1000);
  const d = Math.floor(s / 86400);
  const h = Math.floor((s % 86400) / 3600);
  const m = Math.floor((s % 3600) / 60);
  const sec = s % 60;
  return d > 0 ? `${d}일 ${pad(h)}:${pad(m)}:${pad(sec)}` : `${pad(h)}:${pad(m)}:${pad(sec)}`;
}

// ──────────────────────────────────────
// 공개 페이지 렌더
// ──────────────────────────────────────
function renderPublic() {
  const now = Date.now();
  const nd  = new Date(now);
  $('nowTime').textContent =
    `현재 시각: ${nd.getFullYear()}/${pad(nd.getMonth()+1)}/${pad(nd.getDate())} ` +
    `${pad(nd.getHours())}:${pad(nd.getMinutes())}:${pad(nd.getSeconds())}`;

  if (!allItems.length) {
    $('publicBody').innerHTML = '<tr><td colspan="6" class="empty">등록된 일정이 없습니다</td></tr>';
    $('tabs').innerHTML = '';
    return;
  }

  // 날짜 그룹
  const dayMap = {};
  allItems.forEach(item => {
    const k = getDayKey(item.targetTime);
    if (!dayMap[k]) dayMap[k] = [];
    dayMap[k].push(item);
  });
  const days = Object.keys(dayMap).sort();

  // 탭
  $('tabs').innerHTML = [
    `<button class="tab ${!selectedDay ? 'active' : ''}" onclick="window._day(null)">전체</button>`,
    ...days.map(d => {
      const ts = dayMap[d][0].targetTime;
      return `<button class="tab ${selectedDay === d ? 'active' : ''}" onclick="window._day('${d}')">${fmtDate(ts)}</button>`;
    })
  ].join('');

  // 다음 매치
  let nextItem = null;
  for (const item of allItems) {
    if (item.targetTime > now) { nextItem = item; break; }
  }

  const filtered = selectedDay ? (dayMap[selectedDay] || []) : allItems;
  let html = '';
  let lastDay = null;

  filtered.forEach(item => {
    const dk = getDayKey(item.targetTime);
    if (!selectedDay && dk !== lastDay) {
      lastDay = dk;
      html += `<tr class="day-header"><td colspan="6">📅 ${fmtDate(item.targetTime)}</td></tr>`;
    }
    const ms   = item.targetTime - now;
    const done = ms <= 0;
    const isNext = nextItem && item.id === nextItem.id;
    const cd   = fmtCd(ms);
    const cdCell = cd
      ? `<span class="cd">${cd}</span>${isNext ? '<span class="badge">NEXT</span>' : ''}`
      : `<span style="color:#3a3a3a">✅ 종료</span>`;

    html += `
      <tr class="${done ? 'done' : ''} ${isNext ? 'next' : ''}">
        <td>${fmtDate(item.targetTime)}</td>
        <td>${fmtTime(item.targetTime)}</td>
        <td>${cdCell}</td>
        <td>${item.label || '-'}</td>
        <td>${item.map   || '-'}</td>
        <td>${item.team  || '-'}</td>
      </tr>`;
  });

  $('publicBody').innerHTML = html || '<tr><td colspan="6" class="empty">일정 없음</td></tr>';
}

window._day = function(d) { selectedDay = d; renderPublic(); };

// 1초 타이머
if (ticker) clearInterval(ticker);
ticker = setInterval(renderPublic, 1000);

// ──────────────────────────────────────
// 관리자 목록 렌더
// ──────────────────────────────────────
function renderAdmin() {
  const now = Date.now();
  if (!allItems.length) {
    $('adminBody').innerHTML = '<tr><td colspan="6" class="empty">등록된 일정이 없습니다</td></tr>';
    return;
  }
  $('adminBody').innerHTML = allItems.map(item => {
    const done = item.targetTime <= now;
    return `
      <tr class="${done ? 'done' : ''}">
        <td>${fmtDate(item.targetTime)}</td>
        <td>${fmtTime(item.targetTime)}</td>
        <td>${item.label || '-'}</td>
        <td>${item.map   || '-'}</td>
        <td>${item.team  || '-'}</td>
        <td>
          <button class="btn-sm btn-edit" onclick="window._edit('${item.id}')">수정</button>
          <button class="btn-sm btn-del"  onclick="window._del('${item.id}')">삭제</button>
        </td>
      </tr>`;
  }).join('');
}

// ──────────────────────────────────────
// CRUD
// ──────────────────────────────────────
window.submitSchedule = async function() {
  const timeVal = $('fTime').value;
  const label   = $('fLabel').value.trim();
  const map     = $('fMap').value.trim();
  const team    = $('fTeam').value.trim();

  if (!timeVal || !label) { showToast('날짜/시간과 부분은 필수입니다!', '#e07070'); return; }

  const data = { targetTime: new Date(timeVal).getTime(), label, map, team };

  if (editingId) {
    await set(ref(db, `schedules/${editingId}`), data);
    showToast('✅ 수정 완료!');
  } else {
    await push(ref(db, 'schedules'), data);
    showToast('✅ 추가 완료!');
  }
  resetForm();
};

window._edit = function(id) {
  const item = allItems.find(i => i.id === id);
  if (!item) return;
  editingId = id;
  const d = new Date(item.targetTime);
  $('fTime').value  = `${d.getFullYear()}-${pad(d.getMonth()+1)}-${pad(d.getDate())}T${pad(d.getHours())}:${pad(d.getMinutes())}`;
  $('fLabel').value = item.label || '';
  $('fMap').value   = item.map   || '';
  $('fTeam').value  = item.team  || '';
  $('formTitle').textContent   = '✏️ 일정 수정';
  $('submitBtn').textContent   = '수정 완료';
  window.scrollTo({ top: 0, behavior: 'smooth' });
};

window._del = async function(id) {
  if (!confirm('정말 삭제할까요?')) return;
  await remove(ref(db, `schedules/${id}`));
  showToast('🗑 삭제 완료!', '#e07070');
};

window.resetForm = function() {
  editingId = null;
  $('fTime').value  = '';
  $('fLabel').value = '';
  $('fMap').value   = '';
  $('fTeam').value  = 'Red Team & Green Team';
  $('formTitle').textContent = '➕ 일정 추가';
  $('submitBtn').textContent = '추가';
};

// ──────────────────────────────────────
// 토스트
// ──────────────────────────────────────
function showToast(msg, bg = '#27ae60') {
  const t = $('toast');
  t.textContent = msg;
  t.style.background = bg;
  t.classList.add('show');
  setTimeout(() => t.classList.remove('show'), 2500);
}
