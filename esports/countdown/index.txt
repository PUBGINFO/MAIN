/* 유틸 */
function fmtDate(ts){const d=new Date(ts);const day=['일','월','화','수','목','금','토'][d.getDay()];return `${d.getMonth()+1}/${d.getDate()}(${day})`;}
function fmtTime(ts){const d=new Date(ts);return `${pad(d.getHours())}:${pad(d.getMinutes())}`;}
function getDayKey(ts){const d=new Date(ts);return `${d.getFullYear()}-${pad(d.getMonth()+1)}-${pad(d.getDate())}`;}
function fmtCd(ms){
  if(ms<=0)return null;
  const s=Math.floor(ms/1000),d=Math.floor(s/86400),h=Math.floor((s%86400)/3600),m=Math.floor((s%3600)/60),sec=s%60;
  return d>0?`${d}일 ${pad(h)}:${pad(m)}:${pad(sec)}`:`${pad(h)}:${pad(m)}:${pad(sec)}`;
}

/* 공개 렌더 */
function renderPublic(){
  const now=Date.now(),nd=new Date(now);
  $('nowTime').textContent=`현재 시각: ${nd.getFullYear()}/${pad(nd.getMonth()+1)}/${pad(nd.getDate())} ${pad(nd.getHours())}:${pad(nd.getMinutes())}:${pad(nd.getSeconds())}`;
  if(!allItems.length){$('publicBody').innerHTML='<tr><td colspan="7" class="empty">등록된 일정이 없습니다</td></tr>';$('tabs').innerHTML='';return;}
  const dayMap={};
  allItems.forEach(item=>{const k=getDayKey(item.targetTime);(dayMap[k]=dayMap[k]||[]).push(item);});
  const days=Object.keys(dayMap).sort();
  $('tabs').innerHTML=[
    `<button class="tab ${!selectedDay?'active':''}" onclick="window._day(null)">전체</button>`,
    ...days.map(d=>{const ts=dayMap[d][0].targetTime;return `<button class="tab ${selectedDay===d?'active':''}" onclick="window._day('${d}')">${fmtDate(ts)}</button>`;})
  ].join('');
  let nextItem=null;
  for(const item of allItems){if(item.targetTime>now){nextItem=item;break;}}
  const filtered=selectedDay?(dayMap[selectedDay]||[]):allItems;
  let html='',lastDay=null;
  filtered.forEach(item=>{
    const dk=getDayKey(item.targetTime);
    if(!selectedDay&&dk!==lastDay){lastDay=dk;html+=`<tr class="day-header"><td colspan="7">📅 ${fmtDate(item.targetTime)}</td></tr>`;}
    const ms=item.targetTime-now,done=ms<=0,isNext=nextItem&&item.id===nextItem.id,cd=fmtCd(ms);
    const cdCell=cd?`<span class="cd">${cd}</span>${isNext?'<span class="badge">NEXT</span>':''}`:`<span style="color:var(--txt3)">✅ 종료</span>`;
    const mapCell=item.map?`<span class="map-tag">${item.map}</span>`:'-';
    html+=`<tr class="${done?'done':''} ${isNext?'next':''}">
      <td>${fmtDate(item.targetTime)}</td><td>${fmtTime(item.targetTime)}</td><td>${cdCell}</td>
      <td style="text-align:left;padding-left:16px">${item.title||'-'}</td>
      <td>${item.label||'-'}</td><td>${mapCell}</td><td>${item.team||'-'}</td></tr>`;
  });
  $('publicBody').innerHTML=html||'<tr><td colspan="7" class="empty">일정 없음</td></tr>';
}
window._day=d=>{selectedDay=d;renderPublic();};
if(ticker)clearInterval(ticker);
ticker=setInterval(renderPublic,1000);

function renderAdmin(){
  const now=Date.now();
  if(!allItems.length){$('adminBody').innerHTML='<tr><td colspan="7" class="empty">등록된 일정이 없습니다</td></tr>';return;}
  $('adminBody').innerHTML=allItems.map(item=>{
    const done=item.targetTime<=now;
    return `<tr class="${done?'done':''}">
      <td>${fmtDate(item.targetTime)}</td><td>${fmtTime(item.targetTime)}</td>
      <td style="text-align:left;padding-left:14px">${item.title||'-'}</td>
      <td>${item.label||'-'}</td><td>${item.map||'-'}</td><td>${item.team||'-'}</td>
      <td><button class="btn-sm btn-edit" onclick="window._edit('${item.id}')">수정</button><button class="btn-sm btn-del" onclick="window._del('${item.id}')">삭제</button></td>
    </tr>`;
  }).join('');
}

window.submitSchedule=async function(){
  const timeVal=$('fTime').value,title=$('fTitle').value.trim(),label=$('fLabel').value.trim(),map=$('fMap').value.trim(),team=$('fTeam').value.trim();
  if(!timeVal||!label){showToast('날짜/시간과 라운드는 필수입니다!','#dc2626');return;}
  const data={targetTime:new Date(timeVal).getTime(),title,label,map,team};
  if(editingId){await set(ref(db,`schedules/${editingId}`),data);showToast('✅ 수정 완료!');}
  else{await push(ref(db,'schedules'),data);showToast('✅ 추가 완료!');}
  resetForm();
};
window._edit=function(id){
  const item=allItems.find(i=>i.id===id);if(!item)return;
  editingId=id;
  const d=new Date(item.targetTime);
  $('fTime').value=`${d.getFullYear()}-${pad(d.getMonth()+1)}-${pad(d.getDate())}T${pad(d.getHours())}:${pad(d.getMinutes())}`;
  $('fTitle').value=item.title||'';$('fLabel').value=item.label||'';$('fMap').value=item.map||'';$('fTeam').value=item.team||'';
  $('formTitle').textContent='✏️ 일정 수정';$('submitBtn').textContent='수정 완료';
  window.scrollTo({top:0,behavior:'smooth'});
};
window._del=async function(id){if(!confirm('정말 삭제할까요?'))return;await remove(ref(db,`schedules/${id}`));showToast('🗑 삭제 완료!','#dc2626');};

  $('formTitle').textContent='➕ 일정 추가';$('submitBtn').textContent='추가';
};
