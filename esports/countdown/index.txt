<script type="module">
import { initializeApp } from "https://www.gstatic.com/firebasejs/12.15.0/firebase-app.js";
import { getDatabase, ref, onValue } from "https://www.gstatic.com/firebasejs/12.15.0/firebase-database.js";

  authDomain: "countdown-pubg.firebaseapp.com",
  databaseURL: "https://countdown-pubg-default-rtdb.asia-southeast1.firebasedatabase.app",
  projectId: "countdown-pubg",
  storageBucket: "countdown-pubg.firebasestorage.app",
  messagingSenderId: "474098863700",
  appId: "1:474098863700:web:523b316e14b710970527c1"
};

const app = initializeApp(firebaseConfig);
const db = getDatabase(app);

let allItems = [];
let selectedDay = null;

const pad = n => String(n).padStart(2, '0');

function formatCountdown(ms) {
  if (ms <= 0) return null;
  const s = Math.floor(ms / 1000);
  const d = Math.floor(s / 86400);
  const h = Math.floor((s % 86400) / 3600);
  const m = Math.floor((s % 3600) / 60);
  const sec = s % 60;
  if (d > 0) return `${d}일 ${pad(h)}:${pad(m)}:${pad(sec)}`;
  return `${pad(h)}:${pad(m)}:${pad(sec)}`;
}

function formatDateLabel(ts) {
  const d = new Date(ts);
  const days = ['일','월','화','수','목','금','토'];
  return `${d.getMonth()+1}/${d.getDate()}(${days[d.getDay()]})`;
}

function formatTimeLabel(ts) {
  const d = new Date(ts);
  return `${pad(d.getHours())}:${pad(d.getMinutes())}`;
}

function getDayKey(ts) {
  const d = new Date(ts);
  return `${d.getFullYear()}-${pad(d.getMonth()+1)}-${pad(d.getDate())}`;
}

function render() {
  const now = Date.now();
  const nd = new Date(now);
  document.getElementById('nowTime').textContent =
    `현재 시각: ${nd.getFullYear()}/${pad(nd.getMonth()+1)}/${pad(nd.getDate())} ${pad(nd.getHours())}:${pad(nd.getMinutes())}:${pad(nd.getSeconds())}`;

  if (!allItems.length) {
    document.getElementById('tableBody').innerHTML =
      '<tr><td colspan="6" class="empty">등록된 일정이 없습니다</td></tr>';
    return;
  }

  // 날짜별 그룹
  const dayMap = {};
  allItems.forEach(item => {
    const key = getDayKey(item.targetTime);
    if (!dayMap[key]) dayMap[key] = [];
    dayMap[key].push(item);
  });

  const days = Object.keys(dayMap).sort();

  // 탭 렌더
  const tabsEl = document.getElementById('tabs');
  tabsEl.innerHTML = [
    `<button class="tab ${!selectedDay ? 'active' : ''}" onclick="window._selectDay(null)">전체</button>`,
    ...days.map(d => {
      const ts = dayMap[d][0].targetTime;
      return `<button class="tab ${selectedDay === d ? 'active' : ''}" onclick="window._selectDay('${d}')">${formatDateLabel(ts)}</button>`;
    })
  ].join('');

  // 다음 매치 찾기
  let nextItem = null;
  for (const item of allItems) {
    if (item.targetTime > now) { nextItem = item; break; }
  }

  // 테이블 렌더
  const filtered = selectedDay
    ? (dayMap[selectedDay] || [])
    : allItems;

  let html = '';
  let lastDay = null;

  filtered.forEach(item => {
    const dayKey = getDayKey(item.targetTime);
    if (!selectedDay && dayKey !== lastDay) {
      lastDay = dayKey;
      html += `<tr class="day-header"><td colspan="6">📅 ${formatDateLabel(item.targetTime)}</td></tr>`;
    }

    const ms = item.targetTime - now;
    const done = ms <= 0;
    const isNext = nextItem && item.id === nextItem.id;
    const cd = formatCountdown(ms);

    const cdCell = cd
      ? `<span class="countdown">${cd}</span>${isNext ? '<span class="badge">NEXT</span>' : ''}`
      : `<span class="done-mark">✅ 종료</span>`;

    html += `
      <tr class="${done ? 'done' : ''} ${isNext ? 'next' : ''}">
        <td>${formatDateLabel(item.targetTime)}</td>
        <td>${formatTimeLabel(item.targetTime)}</td>
        <td>${cdCell}</td>
        <td>${item.label || '-'}</td>
        <td>${item.map || '-'}</td>
        <td>${item.team || '-'}</td>
      </tr>`;
  });

  document.getElementById('tableBody').innerHTML = html || '<tr><td colspan="6" class="empty">일정 없음</td></tr>';
}

window._selectDay = function(day) {
  selectedDay = day;
  render();
};

// Firebase 실시간 구독
onValue(ref(db, 'schedules'), snapshot => {
  const data = snapshot.val();
  allItems = data
    ? Object.entries(data)
        .map(([id, v]) => ({ id, ...v }))
        .sort((a, b) => a.targetTime - b.targetTime)
    : [];
  render();
});

setInterval(render, 1000);
